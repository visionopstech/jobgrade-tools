function cors(origin) {
  const ok = ORIGINS.includes(origin);
  return {
    'Access-Control-Allow-Origin': ok ? origin : ORIGINS[0],
    'Access-Control-Allow-Methods': 'POST, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type',
    'Vary': 'Origin',
  };
}

const TOOLS_OK = /^[a-z]{2,20}$/;
const TZ = 12600 * 1000; // Tehran = UTC+3:30
const enc = new TextEncoder();
const hex = (b) => [...new Uint8Array(b)].map((x) => x.toString(16).padStart(2, '0')).join('');
async function hmac(key, msg) {
  const k = await crypto.subtle.importKey('raw', typeof key === 'string' ? enc.encode(key) : key, { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  return crypto.subtle.sign('HMAC', k, enc.encode(msg));
}
// Telegram WebApp initData check: https://core.telegram.org/bots/webapps#validating-data-received-via-the-mini-app
async function tgUser(initData, token) {
  if (!initData || !token) return null;
  const p = new URLSearchParams(initData);
  const hash = p.get('hash');
  if (!hash) return null;
  p.delete('hash');
  const dcs = [...p.entries()].sort((a, b) => (a[0] < b[0] ? -1 : a[0] > b[0] ? 1 : 0)).map(([k, v]) => k + '=' + v).join('\n');
  const secret = await hmac('WebAppData', token);
  if (hex(await hmac(secret, dcs)) !== hash) return null;
  const age = Date.now() / 1000 - parseInt(p.get('auth_date') || '0', 10);
  if (!(age < 60 * 60 * 24 * 30)) return null; // older than 30 days: ask Telegram for fresh data
  try { const u = JSON.parse(p.get('user') || 'null'); return u && u.id ? u : null; } catch { return null; }
}

let schemaReady = false;
async function schema(db) {
  if (schemaReady) return;
  await db.batch([
    db.prepare('CREATE TABLE IF NOT EXISTS users (id TEXT PRIMARY KEY, name TEXT, username TEXT, lang TEXT, premium INTEGER, first_seen INTEGER, last_seen INTEGER, opens INTEGER DEFAULT 0)'),
    db.prepare('CREATE TABLE IF NOT EXISTS progress (uid TEXT, tool TEXT, state TEXT, step INTEGER DEFAULT 0, max_step INTEGER DEFAULT 0, total INTEGER DEFAULT 0, done INTEGER DEFAULT 0, info TEXT, opens INTEGER DEFAULT 0, first INTEGER, updated INTEGER, PRIMARY KEY (uid, tool))'),
    db.prepare('CREATE TABLE IF NOT EXISTS events (id INTEGER PRIMARY KEY AUTOINCREMENT, uid TEXT, tool TEXT, kind TEXT, step INTEGER, ts INTEGER)'),
    db.prepare('CREATE INDEX IF NOT EXISTS ev_ts ON events (ts)'),
  ]);
  schemaReady = true;
}
const clip = (v, n) => (v == null ? null : String(v).slice(0, n));
const int = (v, max) => Math.max(0, Math.min(max, parseInt(v, 10) || 0));

async function sync(req, env, h) {
  if (!env.DB) return json({ error: 'no db' }, 503, h);
  let b;
  try { b = await req.json(); } catch { return json({ error: 'bad json' }, 400, h); }
  const u = await tgUser(String(b.initData || ''), env.BOT_TOKEN);
  if (!u) return json({ error: 'auth' }, 401, h);
  const tool = String(b.tool || '');
  if (!TOOLS_OK.test(tool)) return json({ error: 'bad tool' }, 400, h);
  const db = env.DB;
  await schema(db);
  const uid = String(u.id), now = Date.now();
  const name = clip([u.first_name, u.last_name].filter(Boolean).join(' '), 80);
  const q = [];
  const op = b.op === 'save' ? 'save' : 'open';
  if (op === 'open') {
    q.push(db.prepare('INSERT INTO users (id,name,username,lang,premium,first_seen,last_seen,opens) VALUES (?1,?2,?3,?4,?5,?6,?6,1) ON CONFLICT(id) DO UPDATE SET name=?2, username=?3, lang=?4, premium=?5, last_seen=?6, opens=opens+1')
      .bind(uid, name, clip(u.username, 40), clip(u.language_code, 8), u.is_premium ? 1 : 0, now));
    q.push(db.prepare('INSERT INTO events (uid,tool,kind,step,ts) VALUES (?1,?2,?3,?4,?5)').bind(uid, tool, 'open', null, now));
  } else {
    q.push(db.prepare('UPDATE users SET last_seen=?2 WHERE id=?1').bind(uid, now));
  }
  const prev = await db.prepare('SELECT state, step, updated FROM progress WHERE uid=?1 AND tool=?2').bind(uid, tool).first();
  const step = int(b.step, 999), total = int(b.total, 999), done = b.done ? 1 : 0;
  const info = b.info && typeof b.info === 'object' ? clip(JSON.stringify(b.info), 2000) : null;
  let state = null;
  if (op === 'save' && b.state && typeof b.state === 'object') {
    state = JSON.stringify(b.state);
    if (state.length > 30000) return json({ error: 'too big' }, 413, h);
  }
  if (tool !== 'catalog') {
    // open: count the visit and keep the furthest step; save: also store the tool's saved state
    q.push(db.prepare(`INSERT INTO progress (uid,tool,state,step,max_step,total,done,info,opens,first,updated) VALUES (?1,?2,?3,?4,?4,?5,?6,?7,?8,?9,?10)
      ON CONFLICT(uid,tool) DO UPDATE SET state=COALESCE(?3,state), step=CASE WHEN ?3 IS NULL THEN step ELSE ?4 END, max_step=MAX(max_step,?4), total=?5, done=MAX(done,?6), info=COALESCE(?7,info), opens=opens+?8, updated=CASE WHEN ?3 IS NULL THEN updated ELSE ?9 END`)
      .bind(uid, tool, state, step, total, done, info, op === 'open' ? 1 : 0, now, state ? now : (prev ? prev.updated : null)));
    if (op === 'save' && (!prev || prev.step !== step)) q.push(db.prepare('INSERT INTO events (uid,tool,kind,step,ts) VALUES (?1,?2,?3,?4,?5)').bind(uid, tool, done ? 'done' : 'step', step, now));
  }
  await db.batch(q);
  if (op === 'open') return json({ ok: 1, state: prev && prev.state ? JSON.parse(prev.state) : null, ts: prev ? prev.updated || 0 : 0 }, 200, h);
  return json({ ok: 1, ts: now }, 200, h);
}

async function adminApi(req, env) {
  const H = { 'Cache-Control': 'no-store' };
  const ip = req.headers.get('CF-Connecting-IP') || 'x';
  const rk = 'adm:' + ip + ':' + new Date().toISOString().slice(0, 13);
  const fails = env.LIMITS ? parseInt((await env.LIMITS.get(rk)) || '0', 10) : 0;
  if (fails >= 8) return json({ error: 'wait' }, 429, H);
  let b = {};
  try { b = await req.json(); } catch {}
  const pass = String(b.pass || '');
  const want = String(env.ADMIN_PASS || '');
  let same = want.length > 0 && pass.length === want.length;
  for (let i = 0; i < want.length; i++) same = same && pass.charCodeAt(i) === want.charCodeAt(i);
  if (!same) {
    if (env.LIMITS) await env.LIMITS.put(rk, String(fails + 1), { expirationTtl: 3700 });
    return json({ error: want ? 'pass' : 'nopass' }, 401, H);
  }
  if (!env.DB) return json({ error: 'no db' }, 503, H);
  const db = env.DB;
  await schema(db);
  const now = Date.now();
  const today = Math.floor((now + TZ) / 864e5) * 864e5 - TZ; // Tehran midnight
  const week = today - 6 * 864e5;
  const [kpi, daily, tools, funnel, users, prog] = await db.batch([
    db.prepare('SELECT COUNT(*) total, SUM(last_seen>=?1) act_today, SUM(last_seen>=?2) act_week, SUM(first_seen>=?1) new_today, SUM(first_seen>=?2) new_week FROM users').bind(today, week),
    db.prepare('SELECT CAST((ts+?2)/86400000 AS INTEGER) d, COUNT(DISTINCT uid) n, COUNT(*) opens FROM events WHERE kind=\'open\' AND ts>=?1 GROUP BY d ORDER BY d').bind(today - 13 * 864e5, TZ),
    db.prepare('SELECT tool, COUNT(*) users, SUM(opens) opens, SUM(done) done, SUM(updated>=?1 OR first>=?1) week, MAX(total) total FROM progress GROUP BY tool ORDER BY users DESC').bind(week),
    db.prepare('SELECT tool, max_step s, done, COUNT(*) n FROM progress GROUP BY tool, max_step, done'),
    db.prepare('SELECT id, name, username, lang, premium, first_seen, last_seen, opens FROM users ORDER BY last_seen DESC LIMIT 500'),
    db.prepare('SELECT uid, tool, step, max_step, total, done, info, opens, updated FROM progress'),
  ]);
  const cat = await db.prepare("SELECT COUNT(DISTINCT uid) n, COUNT(*) opens FROM events WHERE tool='catalog' AND kind='open'").first();
  return json({ now, today, kpi: kpi.results[0], daily: daily.results, tools: tools.results, catalog: cat, funnel: funnel.results, users: users.results, progress: prog.results }, 200, H);
}

const json = (o, status, h) => new Response(JSON.stringify(o), { status, headers: { ...h, 'Content-Type': 'application/json; charset=utf-8' } });

export default {
  async fetch(req, env) {
    const origin = req.headers.get('Origin') || '';
    const h = cors(origin);
    if (req.method === 'OPTIONS') return new Response(null, { headers: h });
    const path = new URL(req.url).pathname.replace(/\/+$/, '');
    if (path === '/admin' && req.method === 'GET') return new Response(ADMIN_HTML, { headers: { 'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'no-store', 'X-Frame-Options': 'DENY' } });
    if (path === '/admin/api' && req.method === 'POST') return adminApi(req, env);
    if (path === '/sync') {
      if (!ORIGINS.includes(origin)) return json({ error: 'forbidden' }, 403, h);
      try { return await sync(req, env, h); } catch (e) { return json({ error: 'server', detail: String(e).slice(0, 200) }, 500, h); }
    }
    if (req.method !== 'POST') return json({ error: 'POST only' }, 405, h);
    if (!ORIGINS.includes(origin)) return json({ error: 'forbidden' }, 403, h);

    let body;
    try { body = await req.json(); } catch { return json({ error: 'bad json' }, 400, h); }
    const task = TASKS[body.task];
    const text = String(body.text || '').slice(0, 4000);
    if (!task || !text.trim()) return json({ error: 'bad request' }, 400, h);

    // daily cap per user (Telegram user id if sent, else IP)
    const limit = parseInt(env.DAILY_LIMIT || '10', 10);
    const who = String(body.uid || req.headers.get('CF-Connecting-IP') || 'anon').slice(0, 64);
    const day = new Date().toISOString().slice(0, 10);
    const key = `${day}:${who}`;
    let used = 0;
    if (env.LIMITS) {
      used = parseInt((await env.LIMITS.get(key)) || '0', 10);
      if (used >= limit) return json({ error: 'limit', message: 'سهم امروزت از AI تموم شد. فردا دوباره امتحان کن.' }, 429, h);
    }

    const r = await fetch(env.AI_URL.replace(/\/$/, '') + '/chat/completions', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + env.AI_KEY },
      body: JSON.stringify({
        model: env.AI_MODEL,
        max_tokens: task.max,
        temperature: task.temp ?? 0.6,
        ...(task.json ? { response_format: { type: 'json_object' } } : {}),
        messages: [{ role: 'system', content: task.system }, { role: 'user', content: text }],
      }),
    });
    if (!r.ok) return json({ error: 'ai', status: r.status, detail: (await r.text()).slice(0, 300) }, 502, h);
    const data = await r.json();
    const out = data?.choices?.[0]?.message?.content?.trim() || '';
    if (env.LIMITS) await env.LIMITS.put(key, String(used + 1), { expirationTtl: 60 * 60 * 26 });
    return json({ text: out, left: Math.max(0, limit - used - 1) }, 200, h);
  },
};
